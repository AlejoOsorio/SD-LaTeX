% Comando para secciones preliminares (sin numeración pero en índice)
\newcommand{\prelimchapter}[2]{%

	% Nueva página
	\cleardoublepage

	% Ancla para hyperref
	\phantomsection

	% Entrada en el índice
	\addcontentsline{toc}{chapter}{#1}

	% Encabezados
	\markboth{#1}{#1}

	% Fondo y título personalizado
	\AddToShipoutPictureBG*{%
		\begin{tikzpicture}[remember picture,overlay]

			% Imagen de fondo
			\node[
				inner sep=0pt,
				anchor=north west
			] at (current page.north west)
			{
				\includegraphics[width=\paperwidth]{#2}
			};

			% Caja del título
			\node[
				anchor=north west,
				xshift=1.5cm,
				yshift=-1cm,
				text=black,
				fill=white,
				fill opacity=0.8,
				rounded corners=6pt,
				inner xsep=10pt,
				inner ysep=6pt
			] at (current page.north west)
			{
				\Large\bfseries #1\strut
			};

		\end{tikzpicture}
	}%
}

% Comando para capítulos numerados
\newcommand{\mainchapter}[2]{%

	% Nueva página
	\cleardoublepage

	% Incrementar contador de capítulo y crear ancla
	\refstepcounter{chapter}

	% Entrada en el índice
	\addcontentsline{toc}{chapter}
	{\protect\numberline{\thechapter}#1}

	% Encabezados
	\markboth{\thechapter\quad #1}{\thechapter\quad #1}

	% Fondo y título personalizado
	\AddToShipoutPictureBG*{%
		\begin{tikzpicture}[remember picture,overlay]

			% Imagen de fondo
			\node[
				inner sep=0pt,
				anchor=north west
			] at (current page.north west)
			{
				\includegraphics[width=\paperwidth]{#2}
			};

			% Caja del título
			\node[
				anchor=north west,
				xshift=1.5cm,
				yshift=-1cm,
				text=black,
				fill=white,
				fill opacity=0.8,
				rounded corners=6pt,
				inner xsep=10pt,
				inner ysep=6pt
			] at (current page.north west)
			{
				\Large\bfseries
				\thechapter\quad #1\strut
			};

		\end{tikzpicture}
	}%
}

% Comando para partes del documento (diseño centrado con posición personalizable)
% Uso: \partcover[xpos,ypos]{Título}{imagen.png}
%      \partcover*[xpos,ypos]{Título}{imagen.png} (versión sin numeración de parte)
% xpos: desplazamiento horizontal en puntos desde el centro (positivo = derecha, negativo = izquierda)
% ypos: desplazamiento vertical en puntos desde el centro (positivo = arriba, negativo = abajo)
\makeatletter
\newcommand{\partcover}{\@ifstar{\partcoverstar}{\partcovernormal}}
\newcommand{\partcovernormal}[3][0pt,0pt]{%
	\cleardoublepage%
	\thispagestyle{empty}%
	\refstepcounter{part}%
	\addcontentsline{toc}{part}{\protect\numberline{\thepart}#2}%
	% Procesar las coordenadas del parámetro opcional
	\def\tempcoords{#1}%
	\expandafter\parsecoords\tempcoords\relax%
	\AddToShipoutPictureBG*{%
		\begin{tikzpicture}[remember picture,overlay]
			% Imagen de fondo que ocupa toda la página
			\node[inner sep=0pt] at (current page.center)
			{\includegraphics[width=\paperwidth,height=\paperheight]{#3}};
			% Título en posición personalizable con salto de línea
			\node[
				text=black,
				fill=white, fill opacity=0.9,
				rounded corners=10pt,
				inner xsep=20pt, inner ysep=15pt,
				font=\Huge\bfseries,
				align=center
			] at ([xshift=\xpos,yshift=\ypos]current page.center)
			{PARTE \thepart\\#2};
		\end{tikzpicture}
	}%
	\vspace*{3cm}%
	\phantom{.}%
	\newpage%
}
\newcommand{\partcoverstar}[3][0pt,0pt]{%
	\cleardoublepage%
	\thispagestyle{empty}%
	\phantomsection%
	\addcontentsline{toc}{chapter}{#2}%
	\markboth{#2}{#2}%
	\def\tempcoords{#1}%
	\expandafter\parsecoords\tempcoords\relax%
	\AddToShipoutPictureBG*{%
		\begin{tikzpicture}[remember picture,overlay]
			\node[inner sep=0pt] at (current page.center)
			{\includegraphics[width=\paperwidth,height=\paperheight]{#3}};
			\node[
				text=black,
				fill=white, fill opacity=0.9,
				rounded corners=10pt,
				inner xsep=20pt, inner ysep=15pt,
				font=\Huge\bfseries,
				align=center
			] at ([xshift=\xpos,yshift=\ypos]current page.center)
			{#2};
		\end{tikzpicture}
	}%
	\vspace*{3cm}%
	\phantom{.}%
	\newpage%
}
\makeatother

% Comando auxiliar para procesar coordenadas
\def\parsecoords#1,#2\relax{%
	\def\xpos{#1}%
	\def\ypos{#2}%
}

}

% Custom command with auto-generated label (only 2 parameters)
% Usage: \shadowfig{image_path}{caption_text}
\newcommand{\shadowfig}[2]{
	\begin{figure}[H]
		\centering
		\begin{tcolorbox}[
				colback=white,
				colframe=gray!50,
				boxrule=1pt,
				arc=2pt,
				boxsep=5pt,
				left=3pt,
				right=3pt,
				top=3pt,
				bottom=3pt,
				drop shadow
			]
			\includegraphics[width=0.95\textwidth,keepaspectratio]{#1}
		\end{tcolorbox}
		\caption{#2}
	\end{figure}
}
